from consulta.models import Consulta

-=-==- Recebendo a nossa tabela com todos os seus objetos

consultas = Consulta.objects.all()

-=-==- mostrando todos os objetos do nosso bd, com base em seu titulo e em sua categoria

for consulta in consultas:
    print(f"Processo - {consulta.numero_processo}\nCliente - {consulta.nome_cliente}\nAssunto - {consulta.assunto}\n")

-=-==- Mostrando todos os objetos do nosso bd que possuem a consulta relacionada ao cliente especificado (para verificar de outro cliente, só mudar o nome)

Consulta.objects.filter(nome_cliente__icontains='Pablo')

=-=-=- ordenando a nossa consulta pelos nomes do cliente

Consulta.objects.order_by('nome_cliente')

=-=-=- Retornando o numero de consultas que possuem o status ATIVO

Consulta.objects.filter(ativo=True).count()
